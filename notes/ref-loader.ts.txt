import { readFile } from 'node:fs/promises';
import { join } from 'node:path';
import type { AgentConfig } from '@hphsts/agents';

const FIXTURES_DIR = join(import.meta.dirname, '../../fixtures');

export type DecreeConfig = {
  id: string;
  trigger: 'always' | 'allSettled';
  content: string;
};

export type ScenarioConfig = {
  id: string;
  content: string;
};

export type PresetConfig = {
  id: string;
  name: string;
  actors: string[];
  provocateur?: string;
  head: string;
  scenarios: string[];
  decrees: string[];
};

async function loadJson<TResult>(subdir: string, id: string): Promise<TResult> {
  const path = join(FIXTURES_DIR, subdir, `${id}.json`);
  return JSON.parse(await readFile(path, 'utf-8')) as TResult;
}

export async function loadAgent<TConfig extends AgentConfig = AgentConfig>(
  id: string
): Promise<TConfig> {
  return loadJson<TConfig>('actors', id);
}

export async function loadDecree(id: string): Promise<DecreeConfig> {
  return loadJson<DecreeConfig>('decrees', id);
}

export async function loadScenario(id: string): Promise<ScenarioConfig> {
  return loadJson<ScenarioConfig>('scenarios', id);
}

export async function loadPreset(id: string): Promise<PresetConfig> {
  return loadJson<PresetConfig>('presets', id);
}
