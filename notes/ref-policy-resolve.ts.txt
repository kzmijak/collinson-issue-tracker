import { readdir, readFile, writeFile } from 'node:fs/promises';
import { join, dirname } from 'node:path';
import { z } from 'zod';
import type { Llm } from '@hphsts/agents';
import { validateVault } from '../vault/validate.js';
import { QueryRole } from '../roles/QueryRole.js';
import type { FetchedNode } from '../prompts/SynthesizeAnswerPrompt.js';
import { createActivityLog } from '../activityLog.js';

export const queryInputSchema = {
  question: z.string(),
  context: z.string().optional(),
  onboarding: z.literal('--i-completed-onboarding'),
};

interface QueryArgs {
  question: string;
  context?: string;
}

const MAX_NODE_CONTENT_CHARS = 3000;

export function createQueryTool(vaultRoot: string, llm: Llm) {
  let lastValidatedAt = 0;
  const log = createActivityLog(dirname(vaultRoot));

  async function handler({ question, context }: QueryArgs) {
    const now = Date.now();
    if (now - lastValidatedAt > 15 * 60 * 1000) {
      await validateVault(vaultRoot);
      lastValidatedAt = now;
    }

    const query = new QueryRole(llm);
    const alwaysNodes = await readAlwaysNodes(vaultRoot);
    const alwaysStems = new Set(alwaysNodes.map((n) => n.stem));

    const allFiles = await listKnowledgeFiles(vaultRoot);
    const knownStems = new Set(allFiles.map((f) => f.replace(/\.md$/, '')));

    const index = await readFile(join(vaultRoot, 'index.md'), 'utf-8');
    const rankedStems = await query.rank(index, question, knownStems, 10);

    const rankedNodes = await fetchNodes(vaultRoot, rankedStems, alwaysStems);
    const refNodes = await fetchWikilinkRefs(vaultRoot, rankedNodes, alwaysStems);

    const nodes = dedup([...alwaysNodes, ...rankedNodes, ...refNodes]);

    if (nodes.length === 0) {
      await log.logQuery({ ts: new Date().toISOString(), question, context, stems: [], answer: '' });
      return { content: [{ type: 'text' as const, text: 'No relevant knowledge nodes found.' }] };
    }

    const { answer, gaps } = await query.synthesize(question, context, nodes);

    const stems = nodes.map((n) => n.stem);
    await log.logQuery({ ts: new Date().toISOString(), question, context, stems, answer });
    await writeGapTickets(vaultRoot, question, gaps);

    return { content: [{ type: 'text' as const, text: answer }] };
  }

  return {
    name: 'query_knowledge',
    config: {
      description:
        'Ask the architectural knowledge base a question. Requires onboarding token from orient_wiki. Wiki internally selects relevant nodes and synthesizes an answer.',
      inputSchema: queryInputSchema,
    },
    handler,
  };
}

async function listKnowledgeFiles(vaultRoot: string): Promise<string[]> {
  const knowledgeDir = join(vaultRoot, 'knowledge');
  const entries = await readdir(knowledgeDir, { withFileTypes: true });
  return entries.filter((e) => e.isFile() && e.name.endsWith('.md')).map((e) => e.name);
}

function dedup(nodes: FetchedNode[]): FetchedNode[] {
  const seen = new Set<string>();
  return nodes.filter((n) => {
    if (seen.has(n.stem)) return false;
    seen.add(n.stem);
    return true;
  });
}

async function readAlwaysNodes(vaultRoot: string): Promise<FetchedNode[]> {
  const knowledgeDir = join(vaultRoot, 'knowledge');
  const files = await listKnowledgeFiles(vaultRoot);

  const nodes = await Promise.all(
    files.map(async (file): Promise<FetchedNode | null> => {
      try {
        const content = await readFile(join(knowledgeDir, file), 'utf-8');
        if (!/^scope:\s*always$/m.test(content)) return null;
        return { stem: file.replace(/\.md$/, ''), content: truncate(content) };
      } catch {
        return null;
      }
    })
  );
  return nodes.filter((n): n is FetchedNode => n !== null);
}

async function fetchNodes(
  vaultRoot: string,
  stems: string[],
  excludeStems: Set<string>
): Promise<FetchedNode[]> {
  const knowledgeDir = join(vaultRoot, 'knowledge');

  const nodes = await Promise.all(
    stems.map(async (stem): Promise<FetchedNode | null> => {
      if (excludeStems.has(stem)) return null;
      try {
        const content = await readFile(join(knowledgeDir, `${stem}.md`), 'utf-8');
        return { stem, content: truncate(content) };
      } catch {
        return null;
      }
    })
  );
  return nodes.filter((n): n is FetchedNode => n !== null);
}

async function fetchWikilinkRefs(
  vaultRoot: string,
  nodes: FetchedNode[],
  excludeStems: Set<string>
): Promise<FetchedNode[]> {
  const knowledgeDir = join(vaultRoot, 'knowledge');
  const existingStems = new Set([...excludeStems, ...nodes.map((n) => n.stem)]);

  const refStems = new Set<string>();
  for (const node of nodes) {
    for (const match of node.content.matchAll(/\[\[([^\]]+)\]\]/g)) {
      const stem = match[1];
      if (!existingStems.has(stem)) refStems.add(stem);
    }
  }

  const refs = await Promise.all(
    [...refStems].map(async (stem): Promise<FetchedNode | null> => {
      try {
        const content = await readFile(join(knowledgeDir, `${stem}.md`), 'utf-8');
        return { stem, content: truncate(content) };
      } catch {
        return null;
      }
    })
  );
  return refs.filter((n): n is FetchedNode => n !== null);
}

async function writeGapTickets(vaultRoot: string, question: string, gaps: string[]): Promise<void> {
  if (gaps.length === 0) return;
  const today = new Date().toISOString().slice(0, 10);
  await Promise.all(
    gaps.map(async (gap) => {
      const filename = `gap-${Date.now()}.md`;
      const content = [
        '---',
        'type: gap',
        `detected: ${today}`,
        `question: "${question}"`,
        '---',
        '',
        gap,
        '',
      ].join('\n');
      await writeFile(join(vaultRoot, 'tickets', filename), content, 'utf-8');
    })
  );
}

function truncate(content: string): string {
  return content.length > MAX_NODE_CONTENT_CHARS
    ? content.slice(0, MAX_NODE_CONTENT_CHARS) + '\n… [truncated]'
    : content;
}
